#include <stdio.h>
#include <stdint.h>
#include <string.h>
#include <stdlib.h>
#include <ctype.h>


typedef struct _sAbacus
{
uint8_t number; // 0-255
uint8_t *pUpperRod; // Each element is in {0,1}  
// pUpperRod[]
uint8_t *pLowerRod; // Each element is in {0,1,2,3,4}
// pLowerRodp[]
}sAbacus;

sAbacus * abacus_init( void );
void abacus_free( sAbacus * );
int32_t abacus_set( sAbacus *pA, char *pStr );
int32_t abacus_add( sAbacus *pA, sAbacus b, sAbacus c );
int32_t abacus_del( sAbacus *pA, sAbacus b, sAbacus c );
int32_t abacus_print( sAbacus a );


sAbacus * abacus_init( void )
{
    sAbacus *pA = (sAbacus*)malloc(sizeof(sAbacus));  // 動態分配結構體空間
    if (pA != NULL)  // 檢查分配是否成功
    {
        pA->number = 0;  // 初始化 number 屬性
        pA->pUpperRod = (uint8_t*)calloc(1, sizeof(uint8_t));  // 初始化 pUpperRod 屬性
        pA->pLowerRod = (uint8_t*)calloc(1, sizeof(uint8_t));  // 初始化 pLowerRod 屬性
    }
    return pA;  // 返回指向新建結構體的指針
}


void abacus_free(sAbacus *pA) {
    free(pA->pUpperRod);
    free(pA->pLowerRod);
    free(pA);
}

// Set pA according to pStr
// For example , pStr is "123456789"
// pA -> number = 9;
// pA -> pUpperRod: {0,0,0,0,1,1,1,1,1}
// pA -> pLowerRod: {1,2,3,4,0,1,2,3,4}
// if the length of pStr > 255 or the string contains alphabets or special
//character or NULL, return -1.
// Otherwise , return 0
int32_t abacus_set(sAbacus * pA, char * pStr) {
    

    // 確保 pA 和 pStr 不是 NULL
    if (!pA || !pStr) {
        return -1;
    }

    // 確保字串的長度不超過 255
    int str_len = strlen(pStr);
    if (str_len > 255) {
        return -1;
    }

    // 確保字串中只包含數字字符
    for (int i = 0; i < str_len; i++) {
        if (!isdigit(pStr[i])) {
            return -1;
        }
    }

    // 設置 number 成員變數
    pA->number = str_len;

    // 設置 pUpperRod 和 pLowerRod
    memset(pA->pUpperRod, 0, sizeof(uint8_t) * str_len);
    memset(pA->pLowerRod, 0, sizeof(uint8_t) * str_len); 

    for (int i = 0 ; i <str_len; i++) {
        int j=str_len-i;
        if (pStr[str_len-1-i] < '0' || pStr[str_len-1-i] > '9') {
            return -1;
        }
        int digit = pStr[str_len-1-i] - '0';
        pA->pUpperRod[i] = digit >= 5 ? 1 : 0;
        pA->pLowerRod[i] = digit % 5;
    }

    return 0;
}

// a = b + c
// Successful: return 0; otherwise , return -1
int32_t abacus_add( sAbacus *pA, sAbacus b, sAbacus c ){
    int32_t i;
    uint8_t carry = 0;  // 进位标志
    sAbacus *d = abacus_init();  // 初始化结果为0

    uint8_t len=b.number;
    if(c.number>b.number)len=c.number;
    len++;
    memset(d->pLowerRod, 0, sizeof(uint8_t) * len);
    memset(d->pUpperRod, 0, sizeof(uint8_t) * len);

    // 分别从 b、c、pA 中取出每一位上的数值进行相加，将结果存储到 d 中对应的位上。
    for (i = 0;  i < len-1; i++)
    {
        uint8_t sum = carry;  // 将上一位的进位加入到当前位

        if (i < b.number)
        {
            sum += (b.pUpperRod[i]*5 + b.pLowerRod[i]);
        }
        if (i < c.number)
        {
            sum += (c.pUpperRod[i]*5 + c.pLowerRod[i]);
        }
        //printf("%d\n",sum);
        carry = sum / 10;
        if(carry)sum-=10;  
        d->pUpperRod[i] = (sum>=5);// 取模，得到当前位的值
        d->pLowerRod[i] = sum % 5;

        //printf("%d %d %d %d\n",sum,d->pLowerRod[i],d->pUpperRod[i],carry);

    }

    // 如果最高位有进位，则需要在 d 的最高位上进一位。
    if (carry > 0)
    {
        d->pLowerRod[i] = carry;
        i++;
    }
    if(i>255){
        return -1;
    }
    // 更新 d 的位数并释放 pA 的内存，将 d 赋值给 pA
    
    d->number = i;
    abacus_free(pA);
    *pA = *d;

    //abacus_free(d);

    return 0;
}


// a = b - c
// Successful: return 0; otherwise , return -1
int32_t abacus_del( sAbacus *pA, sAbacus b, sAbacus c ){
    int32_t i;
    int8_t carry = 0;  // 进位标志
    sAbacus *d = abacus_init();  // 初始化结果为0

    uint8_t len=b.number;
    if(c.number>b.number)len=c.number;
    memset(d->pLowerRod, 0, sizeof(uint8_t) * len);
    memset(d->pUpperRod, 0, sizeof(uint8_t) * len);

    // 分别从 b、c、pA 中取出每一位上的数值进行相加，将结果存储到 d 中对应的位上。
    for (i = 0;  i < len; i++)
    {
        int8_t sum = carry;  // 将上一位的借位加入到当前位

        if (i < b.number)
        {
            sum += (b.pUpperRod[i]*5 + b.pLowerRod[i]);
        }
        if (i < c.number)
        {
            sum -= (c.pUpperRod[i]*5 + c.pLowerRod[i]);
        }
        if(sum<0)sum+=10,carry=-1;
        d->pUpperRod[i] = (sum>=5);// 取模，得到当前位的值
        d->pLowerRod[i] = sum % 5;
        //printf("%d %d %d %d\n",sum,d->pLowerRod[i],d->pUpperRod[i],carry);

    }
    //位數重新判定
    for(int i=len-1;i>=0;i--){
        if(d->pLowerRod[i]!=0||d->pUpperRod[i]!=0){
            d->number = i+1;
            break;
        }
    }

    // 如果最高位有进位，则需要在 d 的最高位上进一位。
    if (carry < 0){
        return -1;
    }
    if(i>255){
        return -1;
    }
    // 更新 d 的位数并释放 pA 的内存，将 d 赋值给 pA
    
    
    abacus_free(pA);
    *pA = *d;

    //abacus_free(d);

    return 0;
}
