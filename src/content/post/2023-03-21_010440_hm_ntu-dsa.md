#include<stdio.h>
int m[300][3],ml[3];
int smal(){
    if(ml[0]!=ml[1]){
        return ml[0]>ml[1];
    }else{
        for(int i=ml[0]-1;i>=0;i++){
            if(m[i][1]!=m[i][0]){
                return m[i][0]>m[i][1];    
            }
        }
    } 
    return 0;   
}
int div2(int tar){
    for(int i=ml[tar]-1;i>=0;i--){
        if(m[i][tar]%2)m[i][tar]--,m[i-1][tar]+=10;
        m[i][tar]/=2;
    }
    for(int i=ml[tar]-1;i>=0;i--){
        if(m[i][tar]){//test !=0
            ml[tar]=i+1;
            break;
        }else if(i==0&&m[i][tar]==0){
            ml[tar]=0;
            break;
        }
    }
}
int sub(int sm,int bg){
    for(int i=0;i<ml[bg];i++){
        if(i<ml[sm])m[i][bg]-=m[i][sm];
        if(m[i][bg]<0)m[i+1][bg]--,m[i][bg]+=10;
    }
    for(int i=ml[bg]-1;i>=0;i--){
        if(m[i][bg]){//test !=0
            ml[bg]=i+1;
            break;
        }else if(i==0&&m[i][bg]==0){
            ml[bg]=0;
            break;
        }
    }
}
int mul(int tar){
    int count=0;
    for(int i=ml[tar]-1;i>=0;i--){
        m[i][tar]*=2;
        if(m[i][tar]>=10){
            if(i==ml[tar]-1){
                count=1;
                m[i+1][tar]=1,m[i][tar]-=10;
            }else{
                m[i+1][tar]++,m[i][tar]-=10;
            }
        }
    }
    if(count)ml[tar]++;
}
int BigGcd(){
    int ans =0,sm,bg;
    sm=smal(),bg=!sm;
    while (ml[bg]!=0&&ml[sm]!=0)
    {

        if( m[0][0]%2==0 && m[0][1]%2==0){
            ans++,div2(0),div2(1);
        }else if(m[0][0]%2==0){
            div2(0);
        }else if(m[0][1]%2==0){
            div2(1);
        }
        sm=smal(),bg=!sm;
        sub(sm,bg);
    }
    while(ans--){
        mul(sm);
    }
    return sm;
}
int main(){
    char in[600]={0};
    fgets(in,600,stdin);
    for(int i=0;i<600;i++){//input
        if(in[i]==' '){
            ml[0]=i;
        }else if(in[i]=='\0'||in[i]=='\n'){
            ml[1]=i-ml[0]-1;
            break;
        }
    }
    for(int i=ml[0]-1,j=0;i>=0;i--,j++){
        m[j][0]=(int)in[i]-(int)'0';
    //    printf("%d",m[j][0]);
    }
    //printf(" %d\n",ml[0]);
    for(int i=ml[0]+ml[1],j=0;i>ml[1];i--,j++){
        m[j][1]=(int)in[i]-(int)'0';
    //    printf("%d",m[j][1]);
    }
    //printf(" %d\n",ml[1]);
    int ans=BigGcd();
    for(int i=ml[ans]-1;i>=0;i--){
        printf("%d",m[i][ans]);
    }
    return 0;
}

#include <stdio.h>
#include <stdbool.h>

int n, m;
int row[25][25], col[25][25];
int grid[25][25];

int slove(int level);
void bfs(int depth,int space,int level);
int valid() {
    // 檢查每一列是否合法
    for (int i = 0; i < n; i++) {
        int count = 0, curr = 0;
        bool valid = true;
        for (int j = 0; j < m; j++) {
            if (grid[i][j] == 1) {
                curr++;
            } else {
                if (curr > 0) {
                    if (curr != row[i][count + 1]) {
                        valid = false;
                        break;
                    }
                    count++;
                    curr = 0;
                }
            }
        }
        if (curr > 0 && curr != row[i][count + 1]) {
            valid = false;
        }
        if (!valid) {
            return 0;
        }
    }

    // 檢查每一列是否合法
    for (int j = 0; j < m; j++) {
        int count = 0, curr = 0;
        bool valid = true;
        for (int i = 0; i < n; i++) {
            if (grid[i][j] == 1) {
                curr++;
            } else {
                if (curr > 0) {
                    if (curr != col[j][count + 1]) {
                        valid = false;
                        break;
                    }
                    count++;
                    curr = 0;
                }
            }
        }
        if (curr > 0 && curr != col[j][count + 1]) {
            valid = false;
        }
        if (!valid) {
            return 0;
        }
    }

    return 1;
}

int slove(int level){//直到深度==N
    if(level==5){
        int ans=valid();
        if(ans){
            for(int i=0;i<n;i++){//輸出版面
                for(int j=0;j<m;j++){
                    char c='_';
                    if(grid[i][j])c='o';
                    printf("%c",c);
                }
                printf("\n");
            }
            printf("\n");    
            return 1;
        }
    }else{
        bfs(0,0,level);
    }
}
void bfs(int depth,int space,int level){
    if(depth==row[level][0]){
        slove(level+1);
    }else{
        for(int i=space;i<m;i++){
            if(m-row[level][depth+1]>=i){
                for(int j=0;j<row[level][depth+1];j++){//塗滿
                    grid[level][i+j]=1;
                }
                bfs(depth+1,i+row[level][depth+1]+1,level);
                for(int j=0;j<row[level][depth+1];j++){//清空
                    grid[level][i+j]=0;
                }
            }
        }
    }
}

        col[j][0]=k;
        for (int i = 1; i <= k; i++) {
            scanf("%d", &col[j][i]);
        }
    }

// 解決Nonogram問題
    int ans=slove(0);
return 0;
